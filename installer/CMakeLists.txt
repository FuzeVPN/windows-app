cmake_minimum_required(VERSION 3.20)
project(FuzeVpnInstallerActions LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(fuzevpn_installer_actions SHARED custom_actions.cpp platform_detection.cpp
  ../windows/runner/maintenance_state.cpp ../windows/runner/maintenance_network_state.cpp)
target_compile_features(fuzevpn_installer_actions PRIVATE cxx_std_20)
target_compile_definitions(fuzevpn_installer_actions PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(fuzevpn_installer_actions PRIVATE ../windows/runner)
target_link_libraries(fuzevpn_installer_actions PRIVATE msi version shell32 ole32 advapi32 fwpuclnt iphlpapi ws2_32)
if(MSVC)
  target_compile_options(fuzevpn_installer_actions PRIVATE /W4 /WX /utf-8)
  target_link_options(fuzevpn_installer_actions PRIVATE /DEPENDENTLOADFLAG:0x800)
endif()
